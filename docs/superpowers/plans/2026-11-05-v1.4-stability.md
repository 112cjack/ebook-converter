# V1.4 稳定底座 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地 `docs/改进方案-V1.4.md`：统一 `resolve_engine`、自动转换按源路径、上传流式落盘、本地安装包、无构建拆分 `app.js` + CSS 小打磨；全程测试通过、按任务拆分 commit。

**Architecture:** 方案 A 最小手术。引擎探测下沉到 `ebook_converter.resolve_engine`；自动转换经 `create_path_tasks` 携带既有字段 `src_path`（不 `read_bytes`）；上传流式落在 `POST /api/upload`；安装新增 `mode=local`（multipart → tmp → 复用解压）；前端多 `<script>` 共享 `window.Shudu` 命名空间。不换架构、不引入打包器。

**Tech Stack:** Python 3.10+ / Flask / waitress / 标准库 pathlib·shutil·threading / 原生 JS（无构建）/ pytest / PyInstaller（仅打包，本计划可选重打）。

## Global Constraints

- 全部动态文本经 `textContent` 写入，不拼 HTML（XSS 防线）。
- 服务只绑 127.0.0.1；不扩大任意路径读取 / 转换。
- 中文注释/文案一律 UTF-8；不得用未指定编码的 PowerShell 读写文件。
- 每个任务结束必须全量 `python -m pytest tests -q` 通过后才 commit。
- 自动转换字段沿用既有 `src_path`（不新增 `source_path`）；`upload_id` 对路径任务为 `None`。
- 流式写入落在 `POST /api/upload`；`POST /api/convert` 仍为 JSON 建任务（设计文档「convert」措辞按此澄清）。
- 本地安装包：multipart 上传到 `tmp/` 再解压；失败不静默改走外网下载。
- 前端：无构建多 script；三份 `.spec` 的 `datas` 已打包整个 `webapp/static`，新 js 自动进包，无需改 spec 清单（除非改路径结构）。
- 本计划实现期不改双会话协作铁律（见仓库 `AGENTS.md`）。

```python
def find_portable_engine(base):
    """程序目录下的便携版引擎：engines/calibre 及其一级子目录（兼容官方解压结构）。"""
    root = Path(base) / "engines" / "calibre"
    if not root.is_dir():
        return None
    direct = root / CALIBRE_EXE
    if direct.is_file():
        return direct
    for child in sorted(root.iterdir()):
        candidate = child / CALIBRE_EXE
        if child.is_dir() and candidate.is_file():
            return candidate
    return None


def resolve_engine(base_dir=None, manual_path=None):
    """统一引擎探测：便携版 → 手动路径 → PATH / 常见安装位置。"""
    if base_dir is not None:
        portable = find_portable_engine(base_dir)
        if portable is not None:
            return portable
    manual = str(manual_path or "").strip()
    if manual:
        path = Path(manual)
        if path.is_file():
            return path
    return find_engine()
```

```python
def path_under_roots(path, roots):
    """判断 path 是否落在任一 root 目录之下（含 root 自身文件）。"""
    try:
        resolved = Path(path).resolve()
    except OSError:
        return False
    for root in roots:
        try:
            resolved.relative_to(Path(root).resolve())
            return True
        except (ValueError, OSError):
            continue
    return False
```

```python
def create_path_tasks(self, paths, target_format, output_dir, allowed_roots,
                      overwrite=False, preset="default"):
    if target_format not in ec.OUTPUT_FORMATS:
        raise ValueError(f"不支持的输出格式: {target_format}")
    if preset not in ec.CONVERSION_PRESETS:
        raise ValueError(f"未知的转换预设: {preset}")
    engine = self.resolve_engine()
    if engine is None:
        raise ec.EngineNotFoundError("未检测到 Calibre，请先安装或在设置中指定路径")

    out_dir = Path(output_dir)
    created = []
    with self._lock:
        for raw in paths:
            src = Path(raw)
            now = time.time()
            name = src.name
            fmt_in = src.suffix.lower().lstrip(".")
            task = {
                "id": uuid.uuid4().hex[:12],
                "upload_id": None,
                "name": name,
                "src_path": str(src.resolve()) if src.exists() else str(src),
                "format_in": fmt_in,
                "format_out": target_format,
                "status": QUEUED,
                "progress": 0,
                "message": None,
                "error": None,
                "error_detail": None,
                "output_path": str(out_dir / (src.stem + "." + target_format)),
                "output_dir": str(out_dir),
                "overwrite": overwrite,
                "preset": preset,
                "created_at": now,
                "started_at": None,
                "finished_at": None,
            }
            if not path_under_roots(src, allowed_roots):
                task["status"] = FAILED
                task["error"] = "源文件不在监听目录内"
                task["finished_at"] = now
            elif fmt_in == target_format:
                task["status"] = SKIPPED
                task["message"] = f"源文件已是 {target_format} 格式，无需转换"
                task["finished_at"] = now
            self.tasks[task["id"]] = task
            self.task_order.append(task["id"])
            created.append(task)
            if task["status"] == QUEUED:
                self._queue.put(task["id"])
    if created:
        log.info("按路径创建 %d 个转换任务（目标 %s）", len(created), target_format)
    return created
```

```python
def make_auto_convert_handler(manager, settings_data, base):
    """监听自动转换：按源绝对路径建任务，不整文件读入/复制进 uploads。"""
    def convert(paths, fmt):
        if manager.resolve_engine() is None:
            return 0
        roots = [
            r.get("folder") for r in (settings_data.get("watch_rules") or [])
            if r.get("folder")
        ]
        out_dir = str(settings_data.get("output_dir") or "").strip() or str(Path(base) / "converted")
        try:
            created = manager.create_path_tasks(paths, fmt, Path(out_dir), allowed_roots=roots)
        except ec.EngineNotFoundError:
            return 0
        return len([t for t in created if t["status"] == "queued" or t["status"] == "done"
                    or t["status"] == "converting" or t["status"] == "skipped"])
        # 更简单：return len(created) 即可（与旧行为「创建了多少任务」一致）
    return convert
```

```python
def store_upload(self, filename, data=None, stream=None):
    """保存上传文件。data=bytes 或 stream=可读对象（二选一）。"""
    if (data is None) == (stream is None):
        raise ValueError("store_upload 需要且仅需要 data 或 stream 之一")
    ext = Path(filename).suffix.lower()
    self.upload_dir.mkdir(parents=True, exist_ok=True)
    safe_name, truncated = safe_filename(filename)
    stored = uuid.uuid4().hex[:8] + "_" + safe_name
    stored_path = self.upload_dir / stored
    partial = self.upload_dir / (stored + ".partial")
    size = 0
    try:
        with open(partial, "wb") as fh:
            if data is not None:
                fh.write(data)
                size = len(data)
            else:
                while True:
                    chunk = stream.read(1024 * 64)
                    if not chunk:
                        break
                    fh.write(chunk)
                    size += len(chunk)
        if size == 0:
            partial.unlink(missing_ok=True)
            raise ValueError("文件内容为空")
        partial.replace(stored_path)
    except Exception:
        partial.unlink(missing_ok=True)
        stored_path.unlink(missing_ok=True)
        raise
    record = {
        "id": uuid.uuid4().hex[:12],
        "name": safe_name,
        "stored_name": stored,
        "size": size,
        "format": ext.lstrip("."),
        "removed": False,
        "truncated": truncated,
        "scanned_pdf": ext == ".pdf" and ec.pdf_has_text_layer(stored_path) is False,
    }
    with self._lock:
        self.uploads[record["id"]] = record
        self.upload_order.append(record["id"])
    self.registered_dirs.add(str(self.upload_dir))
    return record
```

```python
MIN_LOCAL_PACKAGE_BYTES = 1 * 1024 * 1024

def start(self, mode, package_path=None):
    if mode not in ("portable", "winget", "local"):
        raise ValueError(f"未知安装方式: {mode}")
    if mode == "local" and not package_path:
        raise ValueError("local 模式需要 package_path")
    with _lock:
        if _state["active"]:
            return False
        msg = {
            "portable": "准备下载",
            "winget": "正在启动 winget",
            "local": "正在准备本地安装包",
        }[mode]
        _state.update(active=True, state="downloading" if mode == "portable" else "installing",
                      progress=0, message=msg)
    threading.Thread(target=self._run, args=(mode, package_path), daemon=True,
                     name="engine-installer").start()
    return True

def _extract_portable(self, installer_exe, target):
    target.mkdir(parents=True, exist_ok=True)
    proc = subprocess.run([str(installer_exe), str(target)], capture_output=True,
                          text=True, timeout=3600, encoding="utf-8", errors="replace")
    return proc

def _run(self, mode, package_path=None):
    try:
        if mode == "winget":
            _set_state(state="installing", message="正在通过 winget 安装 Calibre（官方源）")
            proc = subprocess.run(WINGET_CMD, capture_output=True, text=True,
                                  timeout=3600, encoding="utf-8", errors="replace")
            if proc.returncode != 0:
                tail = (proc.stderr or proc.stdout or "").strip()[-200:]
                _set_state(state="failed", message=tail or "winget 安装失败")
                log.warning("winget 安装失败: %s", tail)
                return
        elif mode == "local":
            dest = Path(package_path)
            if dest.suffix.lower() != ".exe" or not dest.is_file():
                _set_state(state="failed", message="安装包无效或损坏")
                return
            if dest.stat().st_size < MIN_LOCAL_PACKAGE_BYTES:
                _set_state(state="failed", message="安装包无效或损坏（文件过小）")
                return
            _set_state(state="extracting", progress=0, message="正在解压本地安装包到程序目录")
            target = self.base / "engines" / "calibre"
            proc = self._extract_portable(dest, target)
            if proc.returncode != 0:
                tail = (proc.stderr or proc.stdout or "").strip()[-200:]
                _set_state(state="failed", message=tail or "安装包无效或损坏")
                return
            try:
                dest.resolve().relative_to((self.base / "tmp").resolve())
                dest.unlink(missing_ok=True)
            except ValueError:
                pass
        else:
            # portable（与现网一致：下载后再解压）
            dest = self.base / "tmp" / "calibre-portable-installer.exe"
            dest.parent.mkdir(parents=True, exist_ok=True)
            _set_state(state="downloading", message="正在下载便携版引擎（约 100+ MB）")
            _download(PORTABLE_URL, dest, lambda pct: _set_state(progress=pct))
            log.info("便携版安装器下载完成：%s", dest)
            _set_state(state="extracting", progress=100, message="正在解压到程序目录")
            target = self.base / "engines" / "calibre"
            proc = self._extract_portable(dest, target)
            if proc.returncode != 0:
                tail = (proc.stderr or proc.stdout or "").strip()[-200:]
                _set_state(state="failed", message=tail or "便携版解压失败")
                log.warning("便携版解压失败: %s", tail)
                return
            dest.unlink(missing_ok=True)
        _set_state(state="done", progress=100, message="安装完成，正在重新检测")
        log.info("引擎安装完成（%s）", mode)
    except Exception as exc:
        _set_state(state="failed",
                   message=f"安装失败：{exc}。可改用本地安装包、winget 或到官网手动下载。")
        log.warning("引擎安装失败: %s", exc)
        return
    finally:
        _set_state(active=False)
```

```python
def install_engine():
    base = current_app.extensions["base_dir"]
    if request.content_type and "multipart/form-data" in request.content_type:
        mode = str(request.form.get("mode") or "local")
        if mode != "local":

        dest.parent.mkdir(parents=True, exist_ok=True)

        started = installer.EngineInstaller(base).start("local", package_path=str(dest))
    else:
        body = request.get_json(silent=True) or {}
        mode = str(body.get("mode") or "portable")
```
