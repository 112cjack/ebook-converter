```python
MAX_FILENAME_STEM = 80  # 主名截断阈值，避免 tmp 路径 + 输出路径超出 Windows MAX_PATH

def safe_filename(name):
    """清洗上传文件名：去路径成分与 Windows 非法字符，超长主名截断，保留中文与扩展名。"""
    name = Path(str(name).replace("\\", "/")).name
    cleaned = "".join(ch for ch in name if ch.isprintable() and ch not in '<>:"|?*')
    cleaned = cleaned.strip() or "未命名"
    stem, dot, ext = cleaned.rpartition(".")
    truncated = False
    if dot and 0 < len(ext) <= 10:
        if len(stem) > MAX_FILENAME_STEM:
            cleaned, truncated = stem[:MAX_FILENAME_STEM] + dot + ext, True
    elif len(cleaned) > MAX_FILENAME_STEM:
        cleaned, truncated = cleaned[:MAX_FILENAME_STEM], True
    return cleaned, truncated
```

```python
def list_drives():
    if sys.platform == "win32":
        return [f"{c}:" for c in string.ascii_uppercase if Path(f"{c}:/").exists()]
    return ["/"]

@bp.get("/api/browse")
def browse():
    raw = str(request.args.get("path") or "").strip()
    if not raw:
        return jsonify({"current": None, "parent": None,
                        "entries": [{"name": d, "path": d + ("\\" if sys.platform == "win32" else "")}
                                    for d in list_drives()]})
    path = Path(raw)
    if not path.is_absolute():
        return jsonify({"error": "路径必须是绝对路径"}), 400
    if not path.is_dir():
        return jsonify({"error": "目录不存在"}), 400
    entries = []
    try:
        for child in sorted(path.iterdir(), key=lambda p: p.name.lower()):
            if child.name.startswith((".", "$")):
                continue
            if child.is_dir():
                entries.append({"name": child.name, "path": str(child)})
    except OSError as exc:
        return jsonify({"error": f"无法读取目录: {exc}"}), 400
    parent = str(path.parent) if path.parent != path else None
    return jsonify({"current": str(path), "parent": parent, "entries": entries})
```

```python
# -*- coding: utf-8 -*-
"""引擎安装器：官方便携版下载解压 / winget 安装，后台线程执行，状态可轮询。"""
import subprocess
import threading
import urllib.request
from pathlib import Path

PORTABLE_URL = "https://calibre-ebook.com/dist/portable"  # 实施时验证
WINGET_CMD = ["winget", "install", "--id", "calibre.calibre", "-e",
              "--accept-source-agreements", "--accept-package-agreements"]

_state = {"active": False, "state": "idle", "progress": 0, "message": ""}
_lock = threading.Lock()

def install_state():
    with _lock:
        return dict(_state)

def _set_state(**kw):
    with _lock:
        _state.update(kw)

def _download(url, dest, progress_cb=None):
    req = urllib.request.Request(url, headers={"User-Agent": "ebook-converter/1.1"})
    with urllib.request.urlopen(req, timeout=60) as resp, open(dest, "wb") as fh:
        total = int(resp.headers.get("Content-Length") or 0)
        done = 0
        while True:
            chunk = resp.read(1 << 16)
            if not chunk:
                break
            fh.write(chunk)
            done += len(chunk)
            if progress_cb and total:
                progress_cb(min(100, done * 100 // total))

class EngineInstaller:
    def __init__(self, base):
        self.base = Path(base)

    def start(self, mode):
        with _lock:
            if _state["active"]:
                return False
            _state.update(active=True, state="downloading", progress=0, message="准备下载")
        threading.Thread(target=self._run, args=(mode,), daemon=True).start()
        return True

    def wait_done(self, timeout=600):
        import time
        deadline = time.time() + timeout
        while time.time() < deadline and _state["active"]:
            time.sleep(0.05)
        return not _state["active"]

    def _run(self, mode):
        try:
            if mode == "winget":
                _set_state(state="installing", message="正在通过 winget 安装 Calibre…")
                proc = subprocess.run(WINGET_CMD, capture_output=True, text=True, timeout=3600)
                if proc.returncode != 0:
                    _set_state(state="failed", message=(proc.stderr or "").strip()[-200:] or "winget 安装失败")
                    return
            else:
                dest = self.base / "tmp" / "calibre-portable-installer.exe"
                _set_state(state="downloading", message="正在下载便携版引擎（约 100+ MB）")
                _download(PORTABLE_URL, dest, lambda pct: _set_state(progress=pct))
                _set_state(state="extracting", progress=100, message="正在解压到程序目录")
                target = self.base / "engines" / "calibre"
                proc = subprocess.run([str(dest), str(target)], capture_output=True, text=True, timeout=3600)
                if proc.returncode != 0:
                    _set_state(state="failed", message=(proc.stderr or "").strip()[-200:] or "便携版解压失败")
                    return
            _set_state(state="done", active=False, message="安装完成，请重新检测")
        except Exception as exc:
            _set_state(state="failed", active=False, message=f"安装失败: {exc}")
            return
        _set_state(active=False)
```

```python
@bp.post("/api/create-shortcut")
def create_shortcut():
    if not getattr(sys, "frozen", False):
        return jsonify({"error": "仅打包版支持创建桌面快捷方式，脚本版请使用 start.bat"}), 400
    target = Path(sys.executable)
    ps = (
        "$ws = New-Object -ComObject WScript.Shell; "
        "$lnk = [IO.Path]::Combine([Environment]::GetFolderPath('Desktop'), '电子书转换器.lnk'); "
        f"$s = $ws.CreateShortcut($lnk); "
        f"$s.TargetPath = '{str(target).replace(chr(39), chr(39)*2)}'; "
        f"$s.WorkingDirectory = '{str(target.parent).replace(chr(39), chr(39)*2)}'; "
        "$s.Save()"
    )
    try:
        proc = subprocess.run(["powershell", "-NoProfile", "-Command", ps],
                              capture_output=True, text=True, timeout=30)
    except OSError as exc:
        return jsonify({"error": f"创建快捷方式失败: {exc}"}), 500
    if proc.returncode != 0:
        return jsonify({"error": "创建快捷方式失败: " + (proc.stderr or "").strip()[-200:]}), 500
    return jsonify({"ok": True})
```
